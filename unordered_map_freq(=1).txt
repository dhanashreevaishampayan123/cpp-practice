#include <iostream>
#include<unordered_map>
using namespace std;
int main() 
{
   int n;
   cout<<"enter number of elements:";
   cin>>n;
   unordered_map<int,int> freq;
   int key[n];
   cout<<"enter keys:";
   for( int i=0;i<n;i++)
   {
    cin>>key[i];
   }
   for(int i=0;i<n;i++)
   {
       freq[key[i]]++;
   }
   bool found;
   for(int i=0;i<n;i++)
   {
       bool found=false;
       if(freq[key[i]]==1)
       {
           found=true;
           cout<<"output:"<<key[i];
           break;
       }
   }
   if(!found)
   {
       cout<<"no unique element";
   }
   

    return 0;
}

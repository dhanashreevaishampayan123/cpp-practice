#include <iostream>
using namespace std;
int binary_search(int arr[],int n,int key)
{
    int low=0;
    int high=n-1;
    while(low<=high)
    {
        int mid=(low+high)/2;
        if(arr[mid]==key)
        {
            return mid;
        }
        else if(key<arr[mid])
        {
            high=mid-1;
        }
        else
        {
            low=mid+1;
        }
    }
     return -1;
}
int main() {
    int n;
    cout<<"size:";
    cin>>n;
    int array[n];
    cout<<"elements:";
    for(int i=0;i<n;i++)
    {
        cin>>array[i];
    }
    int key;
    cout<<"key:";
    cin>>key;
     int result=binary_search(array,n,key);
    if(result!=-1)
    {
        cout<<"element found at index:"<<result;
    }
    else
    {
        cout<<"element not found.";
    }
    return 0;
}

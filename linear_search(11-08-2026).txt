#include <iostream>
using namespace std;
int linear_search(int arr[],int n,int key)
{
    for(int i=0;i<n;i++)
    {
        if(arr[i]==key)
        {
            return i;
        }
       
    }
     return -1;
}
int main() {
    int n;
    cout<<"size:";
    cin>>n;
    int array[n];
    cout<<"elements:";
    for(int i=0;i<n;i++)
    {
        cin>>array[i];
    }
    int key;
    cout<<"key:";
    cin>>key;
     int result=linear_search(array,n,key);
    if(result!=-1)
    {
        cout<<"element found at index:"<<result;
    }
    else
    {
        cout<<"element not found.";
    }
    return 0;
}
